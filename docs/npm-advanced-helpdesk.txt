# ============================================================
# Nginx Proxy Manager — isi tab "Advanced"
# Proxy Host : helpdesk.rswilliamboothsby.com
# Versi      : mempertahankan sub_filter + location khusus
#              (assets/uploads), PLUS penangkal regex liar:
#              - "location = /sw.js"    (exact match)
#              - "location ^~ /assets/" (prefix ^~)
# ============================================================
#
# KENAPA "= /sw.js" DAN "^~ /assets/"
# Kalau di NPM masih ada location regex ekstensi statis
# (.js/.css/.png/...) — di tab Custom Locations atau di Advanced —
# regex SELALU menang atas location prefix biasa, sehingga
# "location /assets/" tidak pernah kebagian dan /sw.js nyasar.
# Di nginx:
#   - "location = /sw.js"    menang atas SEMUA regex (exact match)
#   - "location ^~ /assets/" menang atas SEMUA regex (prefix ^~)
# Dengan ini /sw.js dan /assets/* dijamin ke upstream yang benar
# tanpa perlu menemukan location liarnya dulu.
#
# WAJIB: blok "location /helpdesk/" tetap ada — sub_filter mengubah
# URL di HTML menjadi /helpdesk/...; tanpa passthrough itu kena
# prefix /helpdesk/ dua kali -> 404.
#
# Verifikasi setelah save — keempat URL ini harus 200:
#   /login.php
#   /manifest.json
#   /sw.js                    (text/javascript)
#   /assets/icons/icon-192.png
#
# CATATAN
# - Ganti 192.168.214.222:8181 bila IP/port app server berubah.
# - Kalau kelak location regex liarnya ketemu dan dihapus
#   (Custom Locations dikosongkan), konfigurasi ini tetap valid.
# ============================================================

# Trafik utama: root domain -> subfolder /helpdesk/ di app server
location / {
    proxy_pass http://192.168.214.222:8181/helpdesk/;
    proxy_set_header Host $host;
    proxy_set_header X-Forwarded-Scheme $scheme;
    proxy_set_header X-Forwarded-Proto $scheme;
    proxy_set_header X-Real-IP $remote_addr;
    proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    proxy_ssl_server_name off;
    proxy_redirect http://192.168.214.222:8181/helpdesk/ /;
    proxy_redirect http://192.168.214.222:8181/ /;
    sub_filter 'href="/assets/' 'href="/helpdesk/assets/';
    sub_filter 'src="/assets/' 'src="/helpdesk/assets/';
    sub_filter 'href="assets/' 'href="/helpdesk/assets/';
    sub_filter 'src="assets/' 'src="/helpdesk/assets/';
    sub_filter 'href="/uploads/' 'href="/helpdesk/uploads/';
    sub_filter 'src="/uploads/' 'src="/helpdesk/uploads/';
    sub_filter 'href="uploads/' 'href="/helpdesk/uploads/';
    sub_filter 'src="uploads/' 'src="/helpdesk/uploads/';
    sub_filter_types text/html text/css text/javascript application/javascript;
    proxy_set_header Accept-Encoding "";
    sub_filter_once off;
}

# Service worker di root domain -> upstream /helpdesk/sw.js
# Exact match: menang atas regex apa pun.
location = /sw.js {
    proxy_pass http://192.168.214.222:8181/helpdesk/sw.js;
    proxy_set_header Host $host;
    proxy_set_header X-Forwarded-Scheme $scheme;
    proxy_set_header X-Forwarded-Proto $scheme;
    proxy_set_header X-Real-IP $remote_addr;
    proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
}

# Assets root (/assets/...) termasuk ikon manifest.
# Prefix ^~: menang atas regex apa pun.
location ^~ /assets/ {
    proxy_pass http://192.168.214.222:8181/helpdesk/assets/;
    proxy_set_header Host $host;
    proxy_set_header X-Forwarded-Scheme $scheme;
    proxy_set_header X-Forwarded-Proto $scheme;
    proxy_set_header X-Real-IP $remote_addr;
    proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
}

# URL hasil sub_filter dan URL lama (/helpdesk/...) diteruskan apa adanya
location /helpdesk/ {
    proxy_pass http://192.168.214.222:8181;
    proxy_set_header Host $host;
    proxy_set_header X-Forwarded-Scheme $scheme;
    proxy_set_header X-Forwarded-Proto $scheme;
    proxy_set_header X-Real-IP $remote_addr;
    proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
}

# Uploads: dua bentuk URL (dengan dan tanpa prefix /helpdesk/)
location /helpdesk/uploads/ {
    proxy_pass http://192.168.214.222:8181/helpdesk/uploads/;
    proxy_set_header Host $host;
    proxy_set_header X-Forwarded-Scheme $scheme;
    proxy_set_header X-Forwarded-Proto $scheme;
    proxy_set_header X-Real-IP $remote_addr;
    proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
}

    proxy_pass http://192.168.214.222:8181/helpdesk/uploads/;
